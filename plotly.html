mean_temps = ny_noaa |>
  filter(!is.na(tmax),
         !is.na(tmin),
         year(date) == 2010) |>
  mutate(date = ymd(date),
        year  = year(date),
        month = month(date),
        day   = day(date),
        tmax = as.numeric(tmax) / 10,
        tmin = as.numeric(tmin) / 10) |>
  group_by(month) |> 
  summarize(
    mean_tmax = mean(tmax, na.rm = TRUE),
    mean_tmin = mean(tmin, na.rm = TRUE),
    .groups = "drop"
  ) |>
  pivot_longer(
    cols = c(mean_tmax, mean_tmin),
    names_to = "temperature_type",
    values_to = "average_temperature"
  ) |> 
  mutate(
    temperature_type = recode(
      temperature_type,
      mean_tmax = "Avg. Maximum Temp",
      mean_tmin = "Avg. Minimum Temp"
  ))

line_plot = ggplot(mean_temps, 
                   aes(x = month, y = average_temperature, color = temperature_type)) +
  geom_line(na.rm = TRUE) +
  scale_x_continuous(breaks = 1:12, labels = month.abb) +
  scale_color_manual(
    values = c("Avg. Maximum Temp" = "red", "Avg. Minimum Temp" = "blue"),
    labels = c(mean_tmax = "Avg. Maximum Temp", mean_tmin = "Avg. Minimum Temp")
  ) +
  labs(
    title = "New York Monthly Average Max and Min temperatures in 2010",
    x = "Month",
    y = "Temperature (°C)",
    color = NULL) +
  theme_minimal()

ggplotly(line_plot)

Monthly temperatures in 2010.

scatter_data = ny_noaa |>
  filter(!is.na(tmax), !is.na(tmin)) |>
  slice_sample(n = 5000) |> 
  mutate(tmax = as.numeric(tmax) / 10,
         tmin = as.numeric(tmin) / 10)

scatter_plot = ggplot(scatter_data, aes(tmin, tmax, text = paste("Date:", date))) +
  geom_point(color = "slateblue", alpha = 0.5, na.rm = TRUE) +
  labs(x = "Minimum temperature (°C)",
       y = "Maximum temperature (°C)") +

snow_data = ny_noaa |>
  mutate(date = lubridate::ymd(date),
    month = factor(
      lubridate::month(date),
      levels = 1:12,
      labels = month.abb
    )
  ) |>
  filter(!is.na(snow), snow > 0) |> 
  slice_sample(n = 5000)

box_plot = ggplot(snow_data, aes(month, snow)) +
  geom_boxplot(fill = "lightblue", na.rm = TRUE) +
  labs(x = "Month", y = "Daily Snow Amount (mm)") +
  theme_minimal()
ggplotly(box_plot)

5000 randomly selected observations, filtered by snowy days and how much it snowed.
